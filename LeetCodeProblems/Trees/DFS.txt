def lowestCA(root,p,q)
	if not root:
		None
	
	if root==p or root==q:
		return root
		
	lca_left=self.lowestCA(root.left,p,q)
	lca_right=self.lowestCA(root.right,p,q)
	
	if lca_left is not None and lca_right is not None:
		return root
		
	if lca_left is not None:
		return lca_left
	
	else:
		return lca_right
